# 系分: 群聊分析工具 (技术版)

## 1. 技术选型与理由

### 1.1 核心决策

| 模块 | 选型 | 理由 |
|------|------|------|
| **运行环境** | OpenClaw Agent | 已有完整 AI Agent 框架，集成 CLI/Browser/Git 等工具 |
| **AI 模型** | MiniMax-M2.1 (默认) | OpenClaw 默认模型，Token 成本低、响应快 |
| **编程语言** | TypeScript / Node.js | 与 OpenClaw 技术栈一致，便于集成 |
| **消息采集** | Python 脚本 + wechaty | wechaty 是成熟的微信机器人 SDK |
| **语义分析** | OpenAI GPT-4o-mini | 高性价比，API 稳定，语义理解能力强 |
| **数据库** | SQLite | 轻量、零配置、数据可移植，适合单机工具 |
| **报告生成** | Node.js + Puppeteer | 渲染 HTML 生成高质量长图 |

### 1.2 放弃的方案

| 方案 | 放弃理由 |
|------|----------|
| Claude API | OpenClaw 未集成，需额外配置 |
| PostgreSQL | 过重，单机工具不需要分布式能力 |
| 自建消息采集 | 微信 Web 协议不稳定，wechaty 更可靠 |

```typescript
import OpenAI from 'openai';

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,

interface ScoringResult {
  member_id: string;
  avg_score: number;
  quality_level: 'excellent' | 'good' | 'normal' | 'low' | 'very_low';
  analysis: string;
}

/**
 * 批量评分成员消息
 */
export async function batchScoreMessages(
  messages: Message[],
  options: { batchSize?: number; model?: string } = {}
): Promise<ScoringResult[]> {
  const { batchSize = 50, model = 'gpt-4o-mini' } = options;

  const results: ScoringResult[] = [];

  // 分批处理，避免单次请求过大
  for (let i = 0; i < messages.length; i += batchSize) {
    const batch = messages.slice(i, i + batchSize);

    const response = await openai.chat.completions.create({
      model,
      messages: [
        {
          role: 'system',
          content: `你是一个群聊质量分析专家。
评分标准 (每条消息 0-10 分):
- 信息密度 (30%): 实质性内容 vs 废话
- 互动价值 (30%): 推进话题 vs 敷衍
- 知识贡献 (25%): 见解/经验 vs 无价值
- 完整度 (15%): 长度、拒绝单字/表情

输出 JSON 格式:
{"scores": [{"member_id": "xxx", "avg_score": 7.5, "quality_level": "good", "analysis": "..."}]}`,
        },
        {
          role: 'user',
          content: JSON.stringify({
            period: '2026-01',
            messages: batch.map(m => ({
              member_id: m.member_id,
              content: m.content,
              type: m.type
            }))
          })
        }
      ],
      response_format: { type: 'json_object' },
      temperature: 0.3,  // 低温度，保证评分一致性
    });

    const parsed = JSON.parse(response.choices[0].message.content!);
    results.push(...parsed.scores);
  }

  return results;
}
```

### 4.3 微信消息导入 (Python)

```python
# scripts/import_msg.py

"""
微信群聊消息导入脚本
Usage: python import_msg.py --group "群名" --output data/messages.json
"""

import json
import hashlib
from datetime import datetime
from typing import List, Dict, Any
import argparse

from wechaty import Wechaty
from wechaty.plugin.ding_dong import DongPlugin


class MessageCollector(Wechaty):
    """消息采集器"""

    def __init__(self, config: dict):
        super().__init__()
        self.config = config
        self.messages: List[Dict] = []
        self.seen_hashes = set()  # 去重用

    async def on_message(self, msg: Message):
        """消息回调"""
        # 只处理群消息
        if msg.room() is None:
            return

        # 过滤: 只处理指定群
        if msg.room().topic() != self.config['group_name']:
            return

        # 构造消息对象
        msg_data = {
            'msg_id': msg.id,
            'group_id': msg.room().id,
            'group_name': msg.room().topic(),
            'member_id': msg.talker().contact_id,
            'member_name': msg.talker().name,
            'content': msg.text(),
            'type': msg.type(),
            'timestamp': msg.date().isoformat(),
        }

        # 去重: SHA256(content + timestamp + member)
        hash_key = hashlib.sha256(
            f"{msg_data['content']}{msg_data['timestamp']}{msg_data['member_id']}".encode()
        ).hexdigest()

        if hash_key in self.seen_hashes:
            return

        self.seen_hashes.add(hash_key)
        self.messages.append(msg_data)

        # 实时写入，避免内存爆
        if len(self.messages) % 100 == 0:
            self._flush_to_file()

    def _flush_to_file(self):
        """落盘"""
        output_path = self.config['output_file']
        with open(output_path, 'a', encoding='utf-8') as f:
            for msg in self.messages[-100:]:
                f.write(json.dumps(msg, ensure_ascii=False) + '\n')
        self.messages = self.messages[-100:]  # 保留最后 100 条

    async def run(self):
        """启动采集"""
        print(f"🚀 开始采集群聊: {self.config['group_name']}")
        await self.start()


def parse_args():
    parser = argparse.ArgumentParser(description='微信消息导入')
    parser.add_argument('--group', required=True, help='群聊名称')
    parser.add_argument('--output', default='./data/messages.jsonl', help='输出文件')
    parser.add_argument('--wechaty-token', env_var='WECHATY_TOKEN', help='Wechaty Token')
    return parser.parse_args()


if __name__ == '__main__':
    args = parse_args()
    config = {
        'group_name': args.group,
        'output_file': args.output,
        'token': args.wechaty_token
    }

    import asyncio
    asyncio.run(MessageCollector(config).run())
```

```typescript
// src/db/operations.ts

import Database from 'better-sqlite3';
import { config } from '../config.js';

const db = new Database(config.db.path);

// 获取周期内成员评分
export function getMemberScores(groupId: string, periodStart: string, periodEnd: string) {
  const stmt = db.prepare(`
    SELECT 
      s.member_id,
      m.name as member_name,
      s.msg_count,
      s.avg_score,
      s.quality_level,
      s.is_whitelisted
    FROM scores s
    JOIN members m ON s.member_id = m.id AND s.group_id = m.group_id
    WHERE s.group_id = ?
      AND s.period_start = ?
      AND s.period_end = ?
      AND s.is_whitelisted = 0
      AND s.quality_level IN ('low', 'very_low')
    ORDER BY 
      CASE s.quality_level 
        WHEN 'very_low' THEN 1 
        WHEN 'low' THEN 2 
      END,
      s.msg_count ASC
  `);
  
  return stmt.all(groupId, periodStart, periodEnd);
}

// 获取精彩内容
export function getHighlights(groupId: string, periodStart: string, periodEnd: string) {
  const stmt = db.prepare(`
    SELECT 
      category,
      content,
      url,
      member_name,
      timestamp
    FROM highlights
    WHERE group_id = ?
      AND period_start = ?
      AND period_end = ?
    ORDER BY timestamp DESC
    LIMIT 10
  `);
  
  return stmt.all(groupId, periodStart, periodEnd);
}

export function bulkInsertScores(scores: ScoreRecord[]) {
  const stmt = db.prepare(`
    INSERT OR REPLACE INTO scores

    VALUES (?, ?, ?, ?, ?, ?, ?, ?)

        record.group_id,

        record.period_end,
        record.msg_count,
        record.avg_score,
        record.quality_level,
        record.ai_analysis

  });
```

### 6.2 重试策略

```typescript
// src/utils/retry.ts

interface RetryOptions {
  maxAttempts?: number;
  delayMs?: number;
  backoff?: number;
  retryCondition?: (error: any) => boolean;
}

export async function withRetry<T>(
  fn: () => Promise<T>,
  options: RetryOptions = {}
): Promise<T> {
  const {
    maxAttempts = 3,
    delayMs = 1000,
    backoff = 2,
    retryCondition = () => true
  } = options;

  let lastError: any;

  for (let attempt = 1; attempt <= maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (error: any) {
      lastError = error;
      
      if (attempt === maxAttempts) {
        throw error;
      }

      if (!retryCondition(error)) {
        throw error;
      }

      const delay = delayMs * Math.pow(backoff, attempt - 1);
      logger.warn(`尝试 ${attempt}/${maxAttempts} 失败, ${delay}ms 后重试...`);
      
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }

  throw lastError;
}

// 使用示例
const scores = await withRetry(
  () => batchScoreMessages(messages),
  {
    maxAttempts: 3,
    retryCondition: (err) => err.code === 'OPENAI_API_ERROR'
  }
);
```

### 6.3 降级策略

```typescript
// src/core/fallback.ts

/**
 * AI 降级策略
 * 当 AI 不可用时，回退到规则引擎
 */

interface RuleScore {
  score: number;
  level: 'low' | 'normal' | 'good';
}

export function fallbackScoring(messages: Message[]): MemberScore[] {
  const memberScores: Map<string, { total: number; count: number }> = new Map();

  for (const msg of messages) {
    const current = memberScores.get(msg.member_id) || { total: 0, count: 0 };
    current.total += calculateRuleScore(msg);
    current.count += 1;
    memberScores.set(msg.member_id, current);
  }

  // 计算平均分
  return Array.from(memberScores.entries()).map(([id, data]) => ({
    member_id: id,
    avg_score: data.total / data.count,
    quality_level: data.total / data.count < 3 ? 'low' : 'normal'
  }));
}

function calculateRuleScore(msg: Message): number {
  const content = msg.content.trim();
  const length = content.length;

  // 规则: 基础分 + 长度分 + 类型分
  
  // 长度分
  let score = 1;
  if (length < 2) score = 0;  // 单字
  else if (length < 10) score = 2;
  else if (length < 50) score = 5;
  else score = 7;

  // 类型分
  if (msg.type === MessageType.EMOTICON) score = Math.min(score, 1);
  if (msg.type === MessageType.LINK) score += 2;
  if (msg.type === MessageType.IMAGE) score += 1;

  // 关键词加分
  const keywords = ['分享', '经验', '建议', 'github', 'http'];
  if (keywords.some(k => content.toLowerCase().includes(k))) {
    score += 2;
  }

  // 扣分项
  if (/^[哈呵哎哟]{1,3}$/.test(content)) score = 0;  // 水群词

  return Math.min(10, score);
}
```
